\section*{Counting Connected Components}

\begin{algorithm}[H]
    \caption{Count Components}
    \label{alg:union-find}
    \begin{algorithmic}[1]   % the optional argument sets the line‑number spacing
        \REQUIRE A graph $G=(V,E)$
        \ENSURE The number of connected components in $G$.

        \STATE Initialize $S$, a union-find data structure with the vertex set $V$
        \STATE $E' \gets E$
        \WHILE{$E' \neq \emptyset$}
            \STATE $e \gets E'.pop()$
            \STATE $s_1 \gets S.find(e.head)$
            \STATE $s_2 \gets S.find(e.tail)$
            \IF{$s_1 \neq s_2$}
                \STATE $S.union(s_1,s_2)$
            \ENDIF
        \ENDWHILE
        \RETURN $S.size()$
    \end{algorithmic}
\end{algorithm}
